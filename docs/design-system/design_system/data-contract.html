§ 08 · Data Contract

The schema behind every surface

Surfaces are views of this data, not the source. Eight core entities, each typed, each carrying provenance through the state enum. Every value a user sees maps to a field here. Demo fixtures exist for design work — always marked DEMO.

§ 08.1

Trust-state enum

The spine of the contract. Every claim and signal carries one of the six trust states; route steps carry a separate workflow status.

type TrustState =
  | 'sourced'    // backed by a real document + ref
  | 'computed'   // derived from sourced inputs
  | 'assumption' // stated, unverified
  | 'unknown'    // human verification required (dashed)
  | 'conflict'   // sources disagree
  | 'decision';  // arbitrated, frozen (ink, filled)

// route step / blocker workflow status (not a trust state)
type RouteStatus = 'ready' | 'blocked' | 'pending';
§ 08.2

Entities

project · parcel · claim · evidence · route · oppositionSignal · blocker · decision.

interface Project {
  id: string; ref: string;            // AED-2026-0042
  office: string; programme: string;
  parcel: Parcel;
  status: State;                      // 'ready' | 'blocked' | …
  claims: Claim[]; route: Route;
  oppositionSignals: OppositionSignal[];
  blockers: Blocker[]; decisions: Decision[];
  reportIds: string[]; createdAt: string;
}

interface Parcel {
  id: string; ref: string;            // CH-VD-1004-4412
  commune: string; zone: string;
  area_m2: number; rdppfRef?: string;
  geometry?: GeoJSON;
}

interface Claim {
  id: string; subject: string;        // 'hauteur_faite'
  statement: string;
  value?: string | number; unit?: string;
  state: TrustState; confidence: number;   // 0–1
  evidenceIds: string[]; source?: string;
}

interface Evidence {
  id: string;
  kind: 'pdf' | 'extract' | 'plan' | 'email' | 'note';
  ref: string; url?: string; page?: number;
  consultedAt: string; confidence: number;
}

interface Route {                       // permit-readiness path
  steps: { id: string; label: string;
           status: RouteStatus; blockerIds: string[] }[];
}

interface OppositionSignal {
  id: string; kind: string; source: string;
  state: TrustState; confidence: number; note?: string;
}

interface Blocker {
  id: string; label: string;
  severity: 'low' | 'med' | 'high';
  status: RouteStatus;                // 'blocked'
  relatedClaimIds: string[];
}

interface Decision {                    // append-only ledger
  id: string;                         // #0042
  text: string; rationale: string;
  author: string; date: string;
  evidenceIds: string[]; supersedes?: string;
}
